cmake_minimum_required(VERSION 3.22.1)
project(ytlocalsub LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/whisper.cpp/CMakeLists.txt")
    message(FATAL_ERROR "whisper.cpp missing: bash scripts/fetch-whisper.sh")
endif()
option(LOCALSUB_ENABLE_VULKAN "Enable experimental Vulkan inference for Android" OFF)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(LOCALSUB_ENABLE_VULKAN)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
else()
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/whisper.cpp" "${CMAKE_CURRENT_BINARY_DIR}/whisper-build")
add_library(localwhisper SHARED native_bridge.cpp)
target_include_directories(localwhisper PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/whisper.cpp/include"
    "${CMAKE_CURRENT_LIST_DIR}/whisper.cpp/ggml/include")
target_link_libraries(localwhisper PRIVATE whisper android log)
if(LOCALSUB_ENABLE_VULKAN)
    target_compile_definitions(localwhisper PRIVATE LOCALSUB_WITH_VULKAN=1)
endif()
target_compile_options(localwhisper PRIVATE -O3 -fvisibility=hidden)
